  def compute_wall_mid(row):
      # Find the level with max bid volume and max ask volume
      bid_vols = [(row.get(f'bid_volume_{i}', 0) or 0,
                   row.get(f'bid_price_{i}', np.nan)) for i in range(1, 4)]
      ask_vols = [(row.get(f'ask_volume_{i}', 0) or 0,
                   row.get(f'ask_price_{i}', np.nan)) for i in range(1, 4)]
      best_bid = max(bid_vols, key=lambda x: x[0])[1]
      best_ask = min(ask_vols, key=lambda x: x[0])[1]  # min price on ask side
      # Actually want max volume ask level:
      best_ask = max(ask_vols, key=lambda x: x[0])[1]
      if pd.isna(best_bid) or pd.isna(best_ask):
          return np.nan
      return (best_bid + best_ask) / 2

      return pd.DataFrame(results).sort_values('avg_pct_rank')

Cell 5a: Standard library Black-Scholes implementation (no scipy).
  Include the full norm_cdf, bs_call, bs_delta, and implied_vol functions.
  These are the exact implementations from imc_prosperity_context.txt.

  def norm_cdf(x):
      a1,a2,a3,a4,a5 = 0.254829592,-0.284496736,1.421413741,-1.453152027,1.061405429
      p = 0.3275911
      sign = 1 if x >= 0 else -1
      x = abs(x) / (2**0.5)
      t = 1.0 / (1.0 + p * x)
      y = 1.0-(((((a5*t+a4)*t)+a3)*t+a2)*t+a1)*t*np.exp(-x*x)
      return 0.5*(1.0 + sign*y)

  def bs_call(S, K, T, sigma):
      if T <= 0 or sigma <= 0: return max(S - K, 0.0)
      d1 = (np.log(S/K) + 0.5*sigma**2*T) / (sigma*np.sqrt(T))
      d2 = d1 - sigma*np.sqrt(T)
      return float(S*norm_cdf(d1) - K*norm_cdf(d2))

  def bs_delta(S, K, T, sigma):
      if T <= 0 or sigma <= 0: return 1.0 if S > K else 0.0
      d1 = (np.log(S/K) + 0.5*sigma**2*T) / (sigma*np.sqrt(T))
      return norm_cdf(d1)

  def implied_vol(mkt_price, S, K, T, tol=1e-6):
      lo, hi = 1e-6, 5.0
      for _ in range(100):
          mid = (lo + hi) / 2
          if bs_call(S, K, T, mid) < mkt_price: lo = mid
          else: hi = mid
          if hi - lo < tol: return (lo + hi) / 2
      return None

Markdown cell: "Generic backtesting framework. Implement strategy logic as

Cell 7a: Backtester engine.
  Build a class-based backtester that:
    - Iterates over price rows in timestamp order
    - Accepts a strategy function: strategy(row, state) → list of (side, price, qty)
    - Simulates fills: a buy order at price P fills if P >= best_ask_1 at that tick;
      a sell order fills if P <= best_bid_1. This is conservative (assumes you
      need to cross the spread to guarantee a fill).
    - Tracks position, cash, and marks to market at each step using mid_price.
    - Hard position limit: rejects orders that would breach it.
    - Returns a DataFrame with columns: timestamp, position, cash, mtm_pnl,
      and a fills DataFrame.

  class Backtester:
      def __init__(self, position_limit: int, initial_cash: float = 0.0):
          self.position_limit = position_limit
          self.initial_cash = initial_cash

      def run(self, prices_df: pd.DataFrame, strategy_fn,
              product: str) -> tuple[pd.DataFrame, pd.DataFrame]:
          """
          Runs strategy_fn on each row of prices_df for the given product.
          Returns (results_df, fills_df).
          """
          prod = prices_df[prices_df['product'] == product].copy()
          prod = prod.sort_values(['day', 'timestamp']).reset_index(drop=True)
          state = {'position': 0, 'cash': self.initial_cash, 'memory': {}}
          results, fills = [], []

          for _, row in prod.iterrows():
              orders = strategy_fn(row, state)
              for side, price, qty in (orders or []):
                  if side == 'buy':
                      ask = row['ask_price_1']
                      if pd.notna(ask) and price >= ask:
                          new_pos = state['position'] + qty
                          if abs(new_pos) <= self.position_limit:
                              state['position'] = new_pos
                              state['cash'] -= ask * qty
                              fills.append({'timestamp': row['timestamp'],
                                            'day': row['day'], 'side': 'buy',
                                            'price': ask, 'qty': qty})
                  elif side == 'sell':
                      bid = row['bid_price_1']
                      if pd.notna(bid) and price <= bid:
                          new_pos = state['position'] - qty
                          if abs(new_pos) <= self.position_limit:
                              state['position'] = new_pos
                              state['cash'] += bid * qty
                              fills.append({'timestamp': row['timestamp'],
                                            'day': row['day'], 'side': 'sell',
                                            'price': bid, 'qty': qty})
              mtm = state['cash'] + state['position'] * row['mid_price']
              results.append({'timestamp': row['timestamp'], 'day': row['day'],
                              'position': state['position'],
                              'cash': state['cash'], 'mtm_pnl': mtm})

          return pd.DataFrame(results), pd.DataFrame(fills)

Cell 7b: PnL reporting function.
  Given the results_df from Backtester.run(), compute and print:
    - Total final PnL
    - PnL per day breakdown
    - Sharpe ratio (mean daily PnL / std daily PnL * sqrt(252))
    - Max drawdown (peak-to-trough on cumulative MTM PnL)
    - Max position reached (check if hitting limit)
    - Number of fills per day
  Plot the cumulative MTM PnL curve over time, with day boundaries as vertical
  dashed lines.

  def report_backtest(results_df, fills_df, label=""):
      final_pnl = results_df['mtm_pnl'].iloc[-1]
      daily_pnl = results_df.groupby('day')['mtm_pnl'].last().diff().fillna(
                  results_df.groupby('day')['mtm_pnl'].last().iloc[0])
      sharpe = daily_pnl.mean() / daily_pnl.std() * (252**0.5) \
               if daily_pnl.std() > 0 else 0
      cummax = results_df['mtm_pnl'].cummax()
      max_dd = (cummax - results_df['mtm_pnl']).max()
      max_pos = results_df['position'].abs().max()
      print(f"[{label}] Final PnL: {final_pnl:,.0f} | Sharpe: {sharpe:.2f} | "
            f"Max Drawdown: {max_dd:,.0f} | Max Position: {max_pos}")
      # Plot cumulative PnL
      fig, (ax1, ax2) = plt.subplots(2, 1, figsize=(14, 6), sharex=True)
      ax1.plot(results_df['timestamp'], results_df['mtm_pnl'])
      ax1.set_title(f"{label} — Cumulative MTM PnL")
      ax2.plot(results_df['timestamp'], results_df['position'])
      ax2.axhline(0, color='gray', linewidth=0.5)
      ax2.set_title("Position over time")
      plt.tight_layout(); plt.show()
